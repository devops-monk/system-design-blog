{{- /* JSON-LD. Overrides the theme's copy, which read `.Params.images` for the image (never set here, so no article carried one) and emitted a bare "https://twitter.com/" in sameAs because params.twitter is empty. Built with dict/jsonify rather than hand-written JSON so a quote or an em-dash in a title cannot produce invalid markup. */ -}} {{- $sameAs := slice -}} {{- with .Site.Params.github }}{{ $sameAs = $sameAs | append (printf "https://github.com/%s" .) }}{{ end -}} {{- with .Site.Params.linkedin }}{{ $sameAs = $sameAs | append (printf "https://www.linkedin.com/in/%s" .) }}{{ end -}} {{- with .Site.Params.twitter }}{{ $sameAs = $sameAs | append (printf "https://twitter.com/%s" .) }}{{ end -}} {{- $org := dict "@type" "Organization" "name" .Site.Params.publisher "url" .Site.BaseURL "sameAs" $sameAs -}} {{- $img := partial "social-image.html" . -}} {{- /* Articles are TechArticle; About/Contact/Guide are plain pages. */ -}} {{- $node := dict -}} {{- if .IsPage -}} {{- $node = dict "@type" (cond (eq .Type "page") "WebPage" "TechArticle") "headline" .Title "name" .Title "inLanguage" "en-US" "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "description" (.Description | default .Site.Params.description) "image" $img "wordCount" .WordCount "keywords" (.Params.tags | default slice) "datePublished" (.PublishDate.Format "2006-01-02") "dateModified" (.Lastmod.Format "2006-01-02") "author" (dict "@type" "Person" "name" .Site.Params.author) "publisher" $org -}} {{- else -}} {{- $node = dict "@type" "WebSite" "name" .Site.Title "url" .Site.BaseURL "description" .Site.Params.description "image" $img "publisher" $org -}} {{- end -}}
